# =================================================================
# Authentication (Better Auth — server lives in ai-financial-control-backend)
# =================================================================
# Authentication and sessions are handled by Better Auth in the separate
# backend repo. This frontend only needs to know where that API is.
# A user's Better Auth session cookie (set by the backend) gates /dashboard.

# Required: URL of the ai-financial-control-backend API (Fastify, separate repo).

NEXT_PUBLIC_API_URL=http://localhost:4000

# -----------------------------------------------------------------
# OAuth providers (SERVER-ONLY — configured in the BACKEND's .env, never here)
# -----------------------------------------------------------------
# Configure these in ~/ai-financial-control-backend/.env:
#   GOOGLE_CLIENT_ID=             (Google Cloud Console > OAuth client)
#   GOOGLE_CLIENT_SECRET=
#   MICROSOFT_CLIENT_ID=          (Azure portal > App registration)
#   MICROSOFT_CLIENT_SECRET=
# A provider is only enabled when both its ID and secret are set.
# Authorized redirect URIs on the provider side:
#   <BETTER_AUTH_URL>/api/auth/callback/google
#   <BETTER_AUTH_URL>/api/auth/callback/microsoft
# Production note: the backend's FRONTEND_ORIGIN must be set to this app's
# public origin so OAuth callbacks redirect back here and stay in the
# trusted-origins allowlist.


# =================================================================
# Build Configuration
# =================================================================
# Set to 'true' when deploying with Docker or self-hosting on a VPS.
# This enables Next.js standalone output mode for smaller deployments.

# 3. Follow the setup instructions below

SENTRY_AUTH_TOKEN=    #Example: sntrys_************************************

NEXT_PUBLIC_SENTRY_DISABLED= "false"
